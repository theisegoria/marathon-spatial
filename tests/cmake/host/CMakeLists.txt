# A host as small as they come, for the HostContract test: adds the library the way a game would.
cmake_minimum_required(VERSION 3.28)
project(MarathonSpatialHostFixture LANGUAGES C CXX)
enable_testing()
add_subdirectory("${LIBRARY}" marathon-spatial EXCLUDE_FROM_ALL)
foreach(required marathon_spatial_core marathon_spatial_physics marathon_spatial_acoustics marathon_spatial_output marathon_spatial)
    if(NOT TARGET ${required})
        message(FATAL_ERROR "host cannot see ${required}")
    endif()
endforeach()
foreach(absent marathon_spatial_core_tests marathon_spatial_physics_tests marathon_spatial_acoustics_tests orbit_offline)
    if(TARGET ${absent})
        message(FATAL_ERROR "${absent} leaked into the host")
    endif()
endforeach()
get_target_property(needs marathon_spatial_acoustics MANUALLY_ADDED_DEPENDENCIES)
if(NOT needs MATCHES "marathon_spatial_acoustics_metallib")
    message(FATAL_ERROR "linking the acoustics does not build its kernels")
endif()
if(NOT MARATHON_SPATIAL_ACOUSTICS_METALLIB)
    message(FATAL_ERROR "the host is not told where the kernels are")
endif()
# Test tools and BUILD_TESTING are on in this host, and still none of the library's tests may appear in its ctest.
add_executable(host_probe "${CMAKE_CURRENT_SOURCE_DIR}/probe.c")
target_link_libraries(host_probe PRIVATE MarathonSpatial::core)
add_test(NAME host_probe COMMAND host_probe)
